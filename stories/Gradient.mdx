export const GradientCard = ({ token, direction, css, description, demoBackground }) => {
  const [copied, setCopied] = React.useState(false);
  const copy = () => navigator.clipboard.writeText(css).then(() => { setCopied(true); setTimeout(() => setCopied(false), 1500); });
  return (
    <div style={{ display: 'flex', flexDirection: 'column', minWidth: 280, flex: '1 1 280px' }}>
      <div onClick={copy} style={{
        position: 'relative',
        height: 180,
        borderRadius: 12,
        overflow: 'hidden',
        cursor: 'pointer',
        background: demoBackground,
      }}>
        {/* Demo content underneath gradient to show fade effect */}
        <div style={{
          position: 'absolute',
          inset: 0,
          background: 'repeating-linear-gradient(135deg, #E1E2E4 0 12px, #F7F8F9 12px 24px)',
        }} />
        {/* Gradient overlay */}
        <div style={{
          position: 'absolute',
          inset: 0,
          background: css,
        }} />
        {/* Direction indicator */}
        <div style={{
          position: 'absolute',
          top: 12,
          left: 12,
          fontSize: 11,
          fontWeight: 700,
          color: '#212126',
          background: 'rgba(255,255,255,0.85)',
          padding: '3px 8px',
          borderRadius: 6,
        }}>
          {direction}
        </div>
        {copied && (
          <div style={{ position: 'absolute', inset: 0, display: 'flex', alignItems: 'center', justifyContent: 'center', backgroundColor: 'rgba(34,195,188,0.95)', color: '#FFFFFF', fontSize: 13, fontWeight: 600 }}>
            CSS 복사 완료!
          </div>
        )}
      </div>
      <div style={{ marginTop: 8 }}>
        <div style={{ fontSize: 13, fontWeight: 600, color: '#171719', marginBottom: 4 }}>{token}</div>
        <div style={{ fontSize: 11, color: '#70737C', lineHeight: 1.5, marginBottom: 4 }}>{description}</div>
        <code style={{ fontSize: 11, color: '#2907B1', background: '#F0ECFE', padding: '2px 6px', borderRadius: 4, wordBreak: 'break-all' }}>{css}</code>
      </div>
    </div>
  );
};

## 사용 예시 (React Native + expo-linear-gradient)

```typescript
import { LinearGradient } from 'expo-linear-gradient';
import { gradientTokensV2 } from '@/tokens/theme';

function ScrollFade() {
  const g = gradientTokensV2.whTopdown;
  return (
    <LinearGradient
      colors={g.colors}
      locations={g.locations}
      start={g.start}
      end={g.end}
      style={{ position: 'absolute', top: 0, left: 0, right: 0, height: 32 }}
    />
  );
}
```
