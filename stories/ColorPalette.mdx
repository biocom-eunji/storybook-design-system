export const Swatch = ({ token, hex, desc, isKey, warningText }) => {
  const [copied, setCopied] = React.useState(false);
  const isLight = hex === '#FFFFFF' || parseInt(hex.slice(1, 3), 16) > 240;
  const copy = () => navigator.clipboard.writeText(hex).then(() => { setCopied(true); setTimeout(() => setCopied(false), 1500); });
  return (
    <div style={{ display: 'flex', flexDirection: 'column', minWidth: 96, flex: '1 1 96px', position: 'relative' }}>
      <div onClick={copy} style={{ height: 64, borderRadius: 8, backgroundColor: hex, border: isLight ? '1px solid #E1E2E4' : 'none', position: 'relative', overflow: 'hidden', cursor: 'pointer' }}>
        {isKey && (
          <span style={{ position: 'absolute', top: 6, right: 6, fontSize: 10, padding: '2px 6px', borderRadius: 10, background: '#171719', color: '#22C3BC', fontWeight: 700 }}>KEY</span>
        )}
        {copied && <div style={{ position: 'absolute', inset: 0, display: 'flex', alignItems: 'center', justifyContent: 'center', backgroundColor: 'rgba(23,23,25,0.75)', borderRadius: 8 }}><span style={{ color: '#22C3BC', fontSize: 11, fontWeight: 600 }}>복사 완료!</span></div>}
      </div>
      <span style={{ fontSize: 12, fontWeight: 600, color: '#171719', marginTop: 6 }}>{token}</span>
      <span onClick={copy} style={{ fontSize: 11, color: '#70737C', fontFamily: 'monospace', cursor: 'pointer' }}>{hex}</span>
      {desc && <span style={{ fontSize: 11, color: warningText ? '#891010' : '#989BA2', marginTop: 2, lineHeight: 1.4 }}>{desc}</span>}
    </div>
  );
};

export const DimSwatch = ({ token, hex, opacity, rgba, baseDark }) => {
  const [copied, setCopied] = React.useState(false);
  const copy = () => navigator.clipboard.writeText(rgba).then(() => { setCopied(true); setTimeout(() => setCopied(false), 1500); });
  return (
    <div style={{ display: 'flex', flexDirection: 'column', minWidth: 160, flex: '1 1 160px' }}>
      <div onClick={copy} style={{
        position: 'relative',
        height: 96,
        borderRadius: 12,
        overflow: 'hidden',
        cursor: 'pointer',
        backgroundImage: baseDark
          ? 'linear-gradient(135deg, #171719 0%, #46474D 100%)'
          : 'repeating-conic-gradient(#F0F1F2 0% 25%, #FFFFFF 0% 50%) 0 0 / 16px 16px',
      }}>
        <div style={{
          position: 'absolute',
          inset: 0,
          background: rgba,
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          color: baseDark ? '#FFFFFF' : (hex === '#000000' && opacity >= 0.6 ? '#FFFFFF' : '#171719'),
          fontSize: 13,
          fontWeight: 600,
          letterSpacing: 0.2,
        }}>
          {token}
        </div>
        {copied && (
          <div style={{ position: 'absolute', inset: 0, display: 'flex', alignItems: 'center', justifyContent: 'center', backgroundColor: 'rgba(34,195,188,0.95)', color: '#FFFFFF', fontSize: 12, fontWeight: 600 }}>
            복사 완료!
          </div>
        )}
      </div>
      <div style={{ marginTop: 8, fontSize: 11, color: '#70737C', fontFamily: 'monospace', lineHeight: 1.5 }}>
        <div>{hex} · opacity {opacity}</div>
        <div style={{ wordBreak: 'break-all' }}>{rgba}</div>
      </div>
    </div>
  );
};
