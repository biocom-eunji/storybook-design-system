export const ToastContext = React.createContext({ copy: () => {} });

export const ToastProvider = ({ children }) => {
  const [msg, setMsg] = React.useState(null);
  const timer = React.useRef(null);
  const lastCall = React.useRef(0);

  const copy = React.useCallback((text) => {
    const now = Date.now();
    if (now - lastCall.current < 200) return; // debounce 200ms
    lastCall.current = now;
    if (timer.current) clearTimeout(timer.current);
    try {
      navigator.clipboard.writeText(text).then(() => {
        setMsg(text);
        timer.current = setTimeout(() => setMsg(null), 1500);
      });
    } catch {
      setMsg(text);
      timer.current = setTimeout(() => setMsg(null), 1500);
    }
  }, []);

  return (
    <ToastContext.Provider value={{ copy }}>
      {children}
      {msg && (
        <div className="units-toast" role="status" aria-live="polite">
          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="#22C3BC" strokeWidth={3} strokeLinecap="round" strokeLinejoin="round">
            <polyline points="20 6 9 17 4 12" />
          </svg>
          <span>Copied: <code>{msg}</code></span>
        </div>
      )}
    </ToastContext.Provider>
  );
};

export const TokenRow = ({ token, children }) => {
  const { copy } = React.useContext(ToastContext);
  const onKeyDown = (e) => {
    if (e.key === 'Enter' || e.key === ' ') {
      e.preventDefault();
      copy(token);
    }
  };
  return (
    <tr
      className="token-row"
      role="button"
      tabIndex={0}
      aria-label={`Copy token ${token}`}
      onClick={() => copy(token)}
      onKeyDown={onKeyDown}
    >
      {children}
    </tr>
  );
};

export const SizeCell = ({ display }) => {
  const m = typeof display === 'string' ? display.match(/^(.*?)\s*(px)$/) : null;
  if (!m) {

  return (
    <td className="value-cell">
      <span className="value-number">{m[1]}</span>
      <span className="value-unit">{m[2]}</span>
    </td>
  );
};

export const TableOfContents = () => {
  const [active, setActive] = React.useState(tocItems[0].id);
  React.useEffect(() => {
    if (typeof IntersectionObserver === 'undefined') return;
    const observer = new IntersectionObserver(
      (entries) => {
        const visible = entries.filter((e) => e.isIntersecting).sort((a, b) => a.target.getBoundingClientRect().top - b.target.getBoundingClientRect().top);
        if (visible[0]) setActive(visible[0].target.id);
      },
      { rootMargin: '-25% 0px -65% 0px', threshold: 0 }
    );
    tocItems.forEach(({ id }) => {
      const el = document.getElementById(id);
      if (el) observer.observe(el);
    });
    return () => observer.disconnect();
  }, []);
  return (
    <nav className="units-toc" aria-label="섹션 목차">
      <ul>
        {tocItems.map(({ id, label }) => (
          <li key={id}>
            <a
              href={`#${id}`}
              className={active === id ? 'active' : ''}
              tabIndex={0}
            >
              {label}
            </a>
          </li>
        ))}
      </ul>
    </nav>
  );
};
