export const DimSwatch = ({ token, hex, opacity, rgba, baseDark }) => {
  const [copied, setCopied] = React.useState(false);
  const copy = () => navigator.clipboard.writeText(rgba).then(() => { setCopied(true); setTimeout(() => setCopied(false), 1500); });
  return (
    <div style={{ display: 'flex', flexDirection: 'column', minWidth: 160, flex: '1 1 160px' }}>
      <div onClick={copy} style={{
        position: 'relative',
        height: 96,
        borderRadius: 12,
        overflow: 'hidden',
        cursor: 'pointer',
        backgroundImage: baseDark
          ? 'linear-gradient(135deg, #171719 0%, #46474D 100%)'
          : 'repeating-conic-gradient(#F0F1F2 0% 25%, #FFFFFF 0% 50%) 0 0 / 16px 16px',
      }}>
        <div style={{
          position: 'absolute',
          inset: 0,
          background: rgba,
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          color: baseDark ? '#FFFFFF' : (hex === '#000000' && opacity >= 0.6 ? '#FFFFFF' : '#171719'),
          fontSize: 13,
          fontWeight: 600,
          letterSpacing: 0.2,
        }}>
          {token}
        </div>
        {copied && (
          <div style={{ position: 'absolute', inset: 0, display: 'flex', alignItems: 'center', justifyContent: 'center', backgroundColor: 'rgba(34,195,188,0.95)', color: '#FFFFFF', fontSize: 12, fontWeight: 600 }}>
            복사 완료!
          </div>
        )}
      </div>
      <div style={{ marginTop: 8, fontSize: 11, color: '#70737C', fontFamily: 'monospace', lineHeight: 1.5 }}>
        <div>{hex} · opacity {opacity}</div>
        <div style={{ wordBreak: 'break-all' }}>{rgba}</div>
      </div>
    </div>
  );
};

    <div style={{ display: 'flex', flexWrap: 'wrap', gap: 12 }}>{children}</div>
  </div>
);
